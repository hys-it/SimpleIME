# Codex Step Script：Android 最單純注音鍵盤 App
# 目標：建立一個最簡單的 Android 注音輸入法 IME
# 特色：倚天注音鍵盤排列、候選字列、黑底綠字風格、基本可輸入中文
# 技術：Kotlin + Android InputMethodService

你是資深 Android/Kotlin 工程師。請依照以下步驟建立一個 Android Studio 專案。
專案目標是「最單純的注音鍵盤 App」，不是完整商業輸入法。
先做可用版本，不做雲端、不做學習、不做 AI、不做複雜詞頻。

============================================================
STEP 1：建立 Android 專案
============================================================

建立一個新的 Android 專案：

Project name: SimpleIME
Package name: com.lisam.simpleime
Language: Kotlin
Minimum SDK: 23
Target SDK: 最新可用版本

專案必須包含：

1. MainActivity
2. SimpleZhuyinImeService
3. KeyboardView 或自訂 LinearLayout 鍵盤 UI
4. CandidateView 或 RecyclerView 候選字列
5. 一個簡單注音字典表

============================================================
STEP 2：AndroidManifest.xml 設定輸入法服務
============================================================

在 AndroidManifest.xml 加入 InputMethodService：

<service
    android:name=".SimpleZhuyinImeService"
    android:permission="android.permission.BIND_INPUT_METHOD"
    android:exported="true">
    <intent-filter>
        <action android:name="android.view.InputMethod" />
    </intent-filter>
    <meta-data
        android:name="android.view.im"
        android:resource="@xml/method" />
</service>

建立 res/xml/method.xml：

<?xml version="1.0" encoding="utf-8"?>
<input-method xmlns:android="http://schemas.android.com/apk/res/android"
    android:settingsActivity="com.lisam.simpleime.MainActivity"
    android:isDefault="false" />

STEP 4：建立 SimpleZhuyinImeService
============================================================

建立 Kotlin 類別：

class SimpleZhuyinImeService : InputMethodService() {

    private lateinit var rootView: LinearLayout
    private lateinit var composingText: TextView
    private lateinit var candidateRow: LinearLayout

    private var currentZhuyin = StringBuilder()

    override fun onCreateInputView(): View {
        rootView = LinearLayout(this)
        rootView.orientation = LinearLayout.VERTICAL
        rootView.setBackgroundColor(Color.BLACK)

        composingText = TextView(this)
        composingText.setTextColor(Color.GREEN)
        composingText.textSize = 22f
        composingText.setPadding(12, 8, 12, 8)
        composingText.text = ""
        rootView.addView(composingText)

        candidateRow = LinearLayout(this)
        candidateRow.orientation = LinearLayout.HORIZONTAL
        candidateRow.setPadding(8, 4, 8, 4)
        rootView.addView(candidateRow)

        createKeyboard()

        return rootView
    }
}

STEP 5：建立倚天注音鍵盤排列
============================================================

鍵盤使用以下排列。
請用 Button 動態建立，不要先用 XML 寫死。

第一列：ㄅ ㄉ ˇ ˋ ㄓ ˊ ˙ ㄚ ㄞ ㄢ ㄦ
第二列：ㄆ ㄊ ㄍ ㄐ ㄔ ㄗ ㄧ ㄛ ㄟ ㄣ
第三列：ㄇ ㄋ ㄎ ㄑ ㄕ ㄘ ㄨ ㄜ ㄠ ㄤ
第四列：ㄈ ㄌ ㄏ ㄒ ㄖ ㄙ ㄩ ㄝ ㄡ ㄥ
第五列：退格 空白 送出

建立函式：

private fun createKeyboard() {
    val rows = listOf(
        listOf("ㄅ", "ㄉ", "ˇ", "ˋ", "ㄓ", "ˊ", "˙", "ㄚ", "ㄞ", "ㄢ", "ㄦ"),
        listOf("ㄆ", "ㄊ", "ㄍ", "ㄐ", "ㄔ", "ㄗ", "ㄧ", "ㄛ", "ㄟ", "ㄣ"),
        listOf("ㄇ", "ㄋ", "ㄎ", "ㄑ", "ㄕ", "ㄘ", "ㄨ", "ㄜ", "ㄠ", "ㄤ"),
        listOf("ㄈ", "ㄌ", "ㄏ", "ㄒ", "ㄖ", "ㄙ", "ㄩ", "ㄝ", "ㄡ", "ㄥ")
    )

    for (row in rows) {
        val rowLayout = LinearLayout(this)
        rowLayout.orientation = LinearLayout.HORIZONTAL
        rowLayout.gravity = Gravity.CENTER
        rootView.addView(rowLayout)

        for (key in row) {
            val button = Button(this)
            button.text = key
            button.textSize = 20f
            button.setTextColor(Color.GREEN)
            button.setBackgroundColor(Color.rgb(20, 20, 20))
            button.setOnClickListener { onZhuyinKey(key) }
            rowLayout.addView(button, LinearLayout.LayoutParams(0, 64, 1f))
        }
    }

    val controlRow = LinearLayout(this)
    controlRow.orientation = LinearLayout.HORIZONTAL
    rootView.addView(controlRow)

    addControlButton(controlRow, "退格") { onBackspace() }
    addControlButton(controlRow, "空白") { onSpace() }
    addControlButton(controlRow, "送出") { commitCurrentText() }
}

============================================================
STEP 6：處理按鍵輸入
============================================================

建立以下函式：

private fun onZhuyinKey(key: String) {
    currentZhuyin.append(key)
    updateComposingText()
    updateCandidates()
}

private fun updateComposingText() {
    composingText.text = currentZhuyin.toString()
}

private fun onBackspace() {
    if (currentZhuyin.isNotEmpty()) {
        currentZhuyin.deleteCharAt(currentZhuyin.length - 1)
        updateComposingText()
        updateCandidates()
    } else {
        currentInputConnection.deleteSurroundingText(1, 0)
    }
}

private fun onSpace() {
    val candidates = getCandidates(currentZhuyin.toString())
    if (candidates.isNotEmpty()) {
        commitText(candidates[0])
    } else {
        currentInputConnection.commitText(" ", 1)
    }
}

private fun commitCurrentText() {
    if (currentZhuyin.isNotEmpty()) {
        val candidates = getCandidates(currentZhuyin.toString())
        if (candidates.isNotEmpty()) {
            commitText(candidates[0])
        } else {
            currentInputConnection.commitText(currentZhuyin.toString(), 1)
            currentZhuyin.clear()
            updateComposingText()
            updateCandidates()
        }
    }
}

private fun commitText(text: String) {
    currentInputConnection.commitText(text, 1)
    currentZhuyin.clear()
    updateComposingText()
    updateCandidates()
}

    return dictionary[code] ?: emptyList()
}

STEP 8：候選字列
============================================================

建立 updateCandidates()：

private fun updateCandidates() {
    candidateRow.removeAllViews()

    val candidates = getCandidates(currentZhuyin.toString())

    for (word in candidates) {
        val tv = TextView(this)
        tv.text = word
        tv.textSize = 24f
        tv.setTextColor(Color.GREEN)
        tv.setPadding(20, 8, 20, 8)
        tv.setOnClickListener { commitText(word) }
        candidateRow.addView(tv)
    }
}

private fun addControlButton(row: LinearLayout, label: String, action: () -> Unit) {
    val button = Button(this)
    button.text = label
    button.textSize = 18f
    button.setTextColor(Color.GREEN)
    button.setBackgroundColor(Color.rgb(30, 30, 30))
    button.setOnClickListener { action() }
    row.addView(button, LinearLayout.LayoutParams(0, 72, 1f))
}
